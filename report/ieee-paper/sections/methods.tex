\subsection{Graph2Vec}
We begin with Graph2Vec~\cite{narayanangraph2vec}, an unsupervised whole-graph embedding method that learns 
fixed-length vector representations in a manner analogous to Doc2Vec for documents. 
The core idea is to treat each graph as a document and its rooted subgraphs, corresponding to local node neighborhoods, as words. 
Under this formulation, graphs that share similar structural patterns are expected to be mapped to nearby points in the embedding space.

\begin{algorithm}[t]
\footnotesize
\caption{Graph2Vec}
\label{alg:graph2vec}
\begin{algorithmic}[1]
\Require Graph set $\mathcal{G}$, WL depth $D$, dim $\delta$, epochs $e$, lr $\alpha$
\Ensure Graph embeddings $\Phi \in \mathbb{R}^{|\mathcal{G}| \times \delta}$

\State Initialize $\Phi \sim \mathcal{U}(-0.5/\delta, 0.5/\delta)$
\For{$epoch=1$ to $e$}
  \State Shuffle $\mathcal{G}$
  \For{each $G_i \in \mathcal{G}$}
    \For{$d=0$ to $D$}
      \For{each node $n \in G_i$}
        \State $sg \gets \textsc{WLSubgraph}(n,G_i,d)$
        \State Update $\Phi_i$ via Skip-Gram($sg,\alpha$)
      \EndFor
    \EndFor
  \EndFor
\EndFor
\State \Return $\Phi$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\footnotesize
\caption{WLSubgraph}
\label{alg:getwlsubgraph}
\begin{algorithmic}[1]
\Require Node $n$, graph $G=(N,E,\lambda)$, depth $d$
\Ensure Rooted subgraph label $sg$

\State $sg \gets \lambda(n)$
\For{$i=1$ to $d$}
  \State $sg \gets \text{hash}\big(sg \,\|\, \{\lambda(u): u \in \mathcal{N}(n)\}\big)$
\EndFor
\State \Return $sg$
\end{algorithmic}
\end{algorithm}



More specifically, Graph2Vec first applies the Weisfeiler--Lehman (WL) relabeling procedure to extract rooted subgraphs around each node up to a specified depth $D$ (Algorithm~\ref{alg:getwlsubgraph}). 
These rooted subgraphs are then aggregated across all graphs and WL iterations to form a global vocabulary, referred to as a \emph{bag of subgraphs}. 
Finally, a skip-gram model is trained to learn graph-level embeddings by maximizing the likelihood of observing the rooted subgraphs associated with each graph (Algorithm~\ref{alg:graph2vec}).

\subsection{NetLSD}
NetLSD~\cite{tsitsulin2018netlsd} represents a graph using the spectrum of its graph Laplacian, 
capturing global structural information. 
The underlying idea is that the eigenvalues of the Laplacian matrix encode important properties 
of the graph structure. 
Rather than using the raw eigenvalues directly, NetLSD summarizes the spectral information 
through a diffusion process, specifically via the heat kernel (Algorithm ~\ref{alg:netlsd}).

% \end{algorithmic}

\begin{algorithm}[t]
\footnotesize
\caption{NetLSD (Heat Trace Signature)}
\label{alg:netlsd}
\begin{algorithmic}[1]
\Require Graph $G=(V,E)$, times $T=\{t_k\}_{k=1}^K$, Laplacian type $\mathcal{L}$, eigenvalues $m$
\Ensure Graph signature $s \in \mathbb{R}^K$

\State Build adjacency $A$ and degree $D$
\State $L \gets \begin{cases}
I - D^{-1/2}AD^{-1/2}, & \mathcal{L}=\text{norm} \\
D-A, & \mathcal{L}=\text{comb}
\end{cases}$

\State Compute smallest $m$ eigenvalues $\{\lambda_i\}_{i=1}^m$ of $L$
\For{$k=1$ to $K$}
  \State $s_k \gets \sum_{i=1}^m e^{-t_k\lambda_i}$
\EndFor
\State $s \gets \log(s)$ \Comment{optional}
\State $s \gets \textsc{Normalize}(s)$ \Comment{optional}
\State \Return $s$
\end{algorithmic}
\end{algorithm}


A key advantage of NetLSD is its permutation invariance and scalability. 
To improve computational efficiency, the method approximates the Laplacian spectral distribution 
using stochastic trace estimation, thereby avoiding full eigendecomposition. 
As a result, NetLSD maintains stable performance even for large graphs.
\subsection{GIN}
The Graph Isomorphism Network (GIN)~\cite{xu2019gin} is a supervised graph neural network designed to maximize 
the discriminative power of GNNs. 
In GIN, node representations are iteratively updated by aggregating information from neighboring nodes. 
At each layer, node features are updated using a sum aggregation function followed by a multilayer perceptron (MLP), 
which enables the model to distinguish different graph structures effectively (Algorithm~\ref{alg:gin_forward}). 
The resulting node embeddings are then aggregated via a global pooling operation to obtain a graph-level representation. 
This representation is subsequently passed to a classifier and trained end-to-end using labeled data 
(Algorithm~\ref{alg:gin_train}).

In contrast to unsupervised methods, GIN learns task-specific embeddings that are optimized directly for classification performance. 
While this typically leads to higher accuracy, it requires labeled datasets, which are often difficult to obtain, 
and incurs higher computational costs due to iterative training.

\begin{algorithm}[t]
\footnotesize
\caption{GIN Forward Pass}
\label{alg:gin_forward}
\begin{algorithmic}[1]
\Require Graph $G=(V,E)$, node features $\{h_v^{(0)}\}$, layers $L$
\Ensure Graph embedding $z_G$

\For{$k=1$ to $L$}
  \ForAll{$v \in V$}
    \State $h_v^{(k)} \gets \textsc{MLP}^{(k)}\!\Big((1+\epsilon^{(k)})h_v^{(k-1)}
    + \sum_{u\in\mathcal{N}(v)} h_u^{(k-1)}\Big)$
  \EndFor
\EndFor
\State $z_G \gets \textsc{READOUT}(\{h_v^{(L)}\})$
\State \Return $z_G$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\footnotesize
\caption{Training GIN}
\label{alg:gin_train}
\begin{algorithmic}[1]
\Require Dataset $\mathcal{D}$, epochs $E$, lr $\alpha$, batch size $B$
\Ensure Trained parameters $\Theta$

\State Initialize $\Theta$
\For{$epoch=1$ to $E$}
  \For{mini-batch $\mathcal{B}\subset\mathcal{D}$}
    \ForAll{$(G,y)\in\mathcal{B}$}
      \State $z_G \gets \textsc{GINForward}(G)$
      \State $\hat y \gets \textsc{Classifier}(z_G)$
    \EndFor
    \State $\mathcal{L}\!\gets\!\frac{1}{|\mathcal{B}|}\!\sum\ell(\hat y,y)$
    \State Update $\Theta$ (e.g., Adam)
  \EndFor
\EndFor
\State \Return $\Theta$
\end{algorithmic}
\end{algorithm}
